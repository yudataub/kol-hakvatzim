בס"ד

### פסוקי הודיה בגאולה

כֹּה אָמַר יְיָ הִנְנִי שָׁב שְׁבוּת אָהֳלֵי יַעֲקוֹב וּמִשְׁכְּנֹתָיו אֲרַחֵם וְנִבְנְתָה עִיר עַל תִּלָּהּ וְאַרְמוֹן עַל מִשְׁפָּטוֹ יֵשֵׁב: וְיָצָא מֵהֶם תּוֹדָה וְקוֹל מְשַׂחֲקִים וְהִרְבִּתִים וְלֹא יִמְעָטוּ וְהִכְבַּדְתִּים וְלֹא יִצְעָרוּ (ירמיהו ל).

כִּי נִחַם יְיָ צִיּוֹן נִחַם כָּל חָרְבֹתֶיהָ וַיָּשֶׂם מִדְבָּרָהּ כְּעֵדֶן וְעַרְבָתָהּ כְּגַן יְיָ שָׂשׂוֹן וְשִׂמְחָה יִמָּצֵא בָהּ תּוֹדָה וְקוֹל זִמְרָה (ישעיהו נא)

כֹּה אָמַר יְיָ עוֹד יִשָּׁמַע בַּמָּקוֹם הַזֶּה... קוֹל שָׂשׂוֹן וְקוֹל שִׂמְחָה קוֹל חָתָן וְקוֹל כַּלָּה קוֹל אֹמְרִים הוֹדוּ אֶת יְיָ צְבָאוֹת כִּי טוֹב יְיָ כִּי לְעוֹלָם חַסְדּוֹ מְבִאִים תּוֹדָה בֵּית יְיָ כִּי אָשִׁיב אֶת שְׁבוּת הָאָרֶץ כְּבָרִאשֹׁנָה אָמַר יְיָ (ירמיהו לג)

הוֹדוּ לַייָ כִּי טוֹב כִּי לְעוֹלָם חַסְדּוֹ: וְאִמְרוּ הוֹשִׁיעֵנוּ אֱלֹהֵי יִשְׁעֵנוּ וְקַבְּצֵנוּ וְהַצִּילֵנוּ מִן הַגּוֹיִם לְהֹדוֹת לְשֵׁם קָדְשֶׁךָ לְהִשְׁתַּבֵּחַ בִּתְהִלָּתֶךָ (דה"י א טז).

הוֹשִׁיעֵנוּ יְיָ אֱלֹהֵינוּ וְקַבְּצֵנוּ מִן הַגּוֹיִם לְהֹדוֹת לְשֵׁם קָדְשֶׁךָ לְהִשְׁתַּבֵּחַ בִּתְהִלָּתֶךָ (תהלים קו).

תִּכָּתֶב זֹאת לְדוֹר אַחֲרוֹן וְעַם נִבְרָא יְהַלֶּל יָהּ (תהילים קב).

לְכוּ נְרַנְּנָה לַייָ נָרִיעָה לְצוּר יִשְׁעֵנוּ: נְקַדְּמָה פָנָיו בְּתוֹדָה בִּזְמִרוֹת נָרִיעַ לוֹ: כִּי אֵל גָּדוֹל יְיָ וּמֶלֶךְ גָּדוֹל עַל כָּל אֱלֹהִים (תהלים צה).

### פסוקים ללא מפרשים.

מִזְמוֹר לְתוֹדָה הָרִיעוּ לַה' כָּל הָאָרֶץ: עִבְדוּ אֶת ה' בְּשִׂמְחָה בֹּאוּ לְפָנָיו בִּרְנָנָה: דְּעוּ כִּי ה' הוּא אֱלֹקִים הוּא עָשָׂנוּ וְלוֹ אֲנַחְנוּ עַמּוֹ וְצֹאן מַרְעִיתוֹ: בֹּאוּ שְׁעָרָיו בְּתוֹדָה חֲצֵרֹתָיו בִּתְהִלָּה הוֹדוּ לוֹ בָּרֲכוּ שְׁמוֹ: כִּי טוֹב ה' לְעוֹלָם חַסְדּוֹ וְעַד דֹּר וָדֹר אֱמוּנָתוֹ:

1. אוֹדֶה יְיָ בְּכָל לִבִּי אֲסַפְּרָה כָּל נִפְלְאוֹתֶיךָ. (תהלים ט)
2. זַמְּרוּ לַיְיָ חֲסִידָיו וְהוֹדוּ לְזֵכֶר קָדְשׁוֹ. (תהלים ל)
3. לְמַעַן יְזַמֶּרְךָ כָבוֹד וְלֹא יִדֹּם יְיָ אֱלֹהַי לְעוֹלָם אוֹדֶךָּ. (תהלים ל)
4. הוֹדוּ לַיְיָ בְּכִנּוֹר בְּנֵבֶל עָשׂוֹר זַמְּרוּ־לוֹ. (תהלים לג)
5. אֲבָרְכָה אֶת יְיָ בְּכָל־עֵת תָּמִיד תְּהִלָּתוֹ בְּפִי. (תהלים לד)
6. שִׂמְחוּ צַדִּיקִים בַּיְיָ וְהוֹדוּ לְזֵכֶר קָדְשׁוֹ. (תהילים צז)
7. שִׁירוּ־לוֹ זַמְּרוּ־לוֹ שִׂיחוּ בְּכָל־נִפְלְאוֹתָיו. (תהילים קה)
8. יוֹדוּ לַיְיָ חַסְדּוֹ וְנִפְלְאוֹתָיו לִבְנֵי אָדָם. (תהילים קז)
9. אוֹדֶה יְיָ מְאֹד בְּפִי וּבְתוֹךְ רַבִּים אֲהַלְלֶנּוּ. (תהילים קט)
10. הַלְלוּיָהּ אוֹדֶה יְיָ בְּכָל־לֵבָב בְּסוֹד יְשָׁרִים וְעֵדָה. (תהילים קיא)
11. לְךָ־אֶזְבַּח זֶבַח תּוֹדָה וּבְשֵׁם יְיָ אֶקְרָא. (תהילים קטז)
12. פִּתְחוּ־לִי שַׁעֲרֵי־צֶדֶק אָבֹא־בָם אוֹדֶה יָהּ. (תהילים קיח)
13. אוֹדְךָ כִּי עֲנִיתָנִי וַתְּהִי־לִי לִישׁוּעָה. (תהילים קיח)
14. חֲצוֹת־לַיְלָה אָקוּם לְהוֹדוֹת לָךְ עַל מִשְׁפְּטֵי צִדְקֶךָ. (תהילים קיט)
15. יוֹדוּךָ יְיָ כָּל־מַעֲשֶׂיךָ וַחֲסִידֶיךָ יְבָרְכוּכָה. (תהילים קמה)
16. אֲהַלְלָה שֵׁם אֱלֹהִים בְּשִׁיר וַאֲגַדְּלֶנּוּ בְתוֹדָה. (תהלים סט)
17. יִמָּלֵא פִי תְּהִלָּתֶךָ כָּל הַיּוֹם תִּפְאַרְתֶּךָ. (תהילים עא)
18. וַאֲנַחְנוּ עַמְּךָ וְצֹאן מַרְעִיתֶךָ נוֹדֶה לְּךָ לְעוֹלָם לְדֹר וָדֹר נְסַפֵּר תְּהִלָּתֶךָ. (תהילים עט)
19. נְקַדְּמָה פָנָיו בְּתוֹדָה בִּזְמִרוֹת נָרִיעַ לוֹ. (תהילים צה)
20. כֹּל הַנְּשָׁמָה תְּהַלֵּל יָהּ הַלְלוּיָהּ. (תהילים קנ)
21. מִי חָכָם וְיִשְׁמָר אֵלֶּה וְיִתְבּוֹנְנוּ חַסְדֵי יְיָ . (תהילים קז)
22. זִכְרוּ נִפְלְאוֹתָיו אֲשֶׁר עָשָׂה. (תהלים קה)
23. וְתָמִיד נְסַפֵּר תְּהִלָּתֶךָ. וּנְהַלֵּל לְשֵׁם תִּפְאַרְתֶּךָ (שיר היחוד)
24. הוֹשִׁיעֵנוּ יְיָ אֱלֹהֵינוּ וְקַבְּצֵנוּ מִן הַגּוֹיִם לְהֹדוֹת לְשֵׁם קָדְשֶׁךָ לְהִשְׁתַּבֵּחַ בִּתְהִלָּתֶךָ. (תהלים קו).
25. תִּכָּתֶב זֹאת לְדוֹר אַחֲרוֹן וְעַם נִבְרָא יְהַלֶּל יָהּ. (תהילים קב).
26. אֶלֹהִים שִׁיר חָדָשׁ אָשִׁירָה לָּךְ בְּנֵבֶל עָשׂוֹר אֲזַמְּרָה לָּךְ. (תהילים קמד)
27. אָשִׁירָה לַייָ כִּי גָאֹה גָּאָה סוּס וְרֹכְבוֹ רָמָה בַיָּם. (שמות טו)
28. עָזִּי וְזִמְרָת יָהּ וַיְהִי לִי לִישׁוּעָה זֶה אֵלִי וְאַנְוֵהוּ אֱלֹהֵי אָבִי וַאֲרֹמְמֶנְהוּ. (שמות טו)
29. אָנֹכִי לַייָ אָנֹכִי אָשִׁירָה אֲזַמֵּר לַייָ אֱלֹהֵי יִשְׂרָאֵל. (שופטים ה)
30. וַאֲנִי בְּחַסְדְּךָ בָטַחְתִּי יָגֵל לִבִּי בִּישׁוּעָתֶךָ אָשִׁירָה לַייָ כִּי גָמַל עָלָי. (תהלים יג)
31. נָכוֹן לִבִּי אֱלֹהִים נָכוֹן לִבִּי אָשִׁירָה וַאֲזַמֵּרָה. (תהלים נז)
32. עוּרָה כְבוֹדִי עוּרָה הַנֵּבֶל וְכִנּוֹר אָעִירָה שָּׁחַר. (תהלים נז)
33. לְדָוִד מִזְמוֹר חֶסֶד וּמִשְׁפָּט אָשִׁירָה לְךָ יְיָ אֲזַמֵּרָה. (תהלים קא)
34. אָשִׁירָה לַייָ בְּחַיָּי אֲזַמְּרָה לֵאלֹהַי בְּעוֹדִי. יֶעֱרַב עָלָיו שִׂיחִי אָנֹכִי אֶשְׂמַח בַּייָ. (תהלים קד)
35. וַאֲנִי בְּקוֹל תּוֹדָה אֶזְבְּחָה לָּךְ אֲשֶׁר נָדַרְתִּי אֲשַׁלֵּמָה יְשׁוּעָתָה לַייָ. (יונה ב)
36. אוֹדְךָ בְּקָהָל רָב בְּעַם עָצוּם אֲהַלְלֶךָּ. (תהלים לה)
37. אוֹדְךָ לְעוֹלָם כִּי עָשִׂיתָ וַאֲקַוֶּה שִׁמְךָ כִי טוֹב נֶגֶד חֲסִידֶיךָ. (תהלים פרק נב)
38. וְעַתָּה אֱלֹהֵינוּ מוֹדִים אֲנַחְנוּ לָךְ וּמְהַלְלִים לְשֵׁם תִּפְאַרְתֶּךָ. (דה"י א כט)
39. וְקֵרַבְתָּנוּ מַלְכֵּנוּ לְשִׁמְךָ הַגָּדוֹל סֶלָה. בֶּאֱמֶת בְּאַהֲבָה לְהוֹדוֹת לְךָ (יוצר דשחרית)
40. לַשְׁמִעַ בְּקוֹל תּוֹדָה וּלְסַפֵּר כָּל נִפְלְאוֹתֶיךָ. (תהלים כו)
41. כִּי חַסְדְּךָ לְנֶגֶד עֵינָי וְהִתְהַלַּכְתִּי בַּאֲמִתֶּךָ. (תהלים כו).
42. וַאֲנִי אָשִׁיר עֻזֶּךָ וַאֲרַנֵּן לַבֹּקֶר חַסְדֶּךָ כִּי הָיִיתָ מִשְׂגָּב לִי וּמָנוֹס בְּיוֹם צַר לִי. (תהלים נט).
43. עֻזִּי אֵלֶיךָ אֲזַמֵּרָה כִּי אֱלֹהִים מִשְׂגַּבִּי אֱלֹהֵי חַסְדִּי. (תהלים נט)
44. חַסְדֵי יְיָ עוֹלָם אָשִׁירָה לְדֹר וָדֹר אוֹדִיעַ אֱמוּנָתְךָ בְּפִי. (תהלים פט)
45. בָּרֲכִי נַפְשִׁי אֶת יְיָ וְאַל תִּשְׁכְּחִי כָּל גְּמוּלָיו. (תהלים קג)

### פסוקים עם מפרשים.

וְזָכַרְתָּ אֶת כָּל הַדֶּרֶךְ (דברים ח, ב) מצוות עשה מהתורה לזכור החסדים

שעושה עימו הקב"ה ולהתבונן בהם (רבינו יונה (ג,יז) וחרדים (א,כג)).

כֹּל הַנִּקְרָא בִשְׁמִי וְלִכְבוֹדִי בְּרָאתִיו יְצַרְתִּיו אַף עֲשִׂיתִיו: (ישעיהו מג, ז)

רד"ק: צריך האדם להתבונן ביצירת האדם ומתקון איבריו היאך נעשו בחכמה ובתקון פרנסתו מיום הולדו וזאת החכמה מביאה האדם להכיר בוראו ולשבחו ולהודות לשמו **וזהו כבוד האל**.

כִּי טוֹב חַסְדְּךָ מֵחַיִּים שְׂפָתַי יְשַׁבְּחוּנְךָ. (תהלים סג)

מצודות: כי כל טובת חסדך מנתינת החיים הוא למען ישבחונך שפתי בעודי כי לולא זאת למה לי חיים.

אלשיך: אחד מהחסדים הגדולים וטובים אשר עשה הוא ית' עמנו, הוא שהחשיבנו להללו בפינו, והוא מאמר המשורר כי טוב הוא חסדך אשר אתה עושה עמי יותר מתתך לי חיים, **במה שזכיתני ששפתי ישבחונך**.

אוֹדֶה ה' בְּכָל לִבִּי אֲסַפְּרָה כָּל נִפְלְאוֹתֶיךָ (תהלים ט, ב)

מלבי"ם: לא כדרך בני אדם שלא יספרו רק נסים הגלוים, אבל אנכי נפקחו עיני לדעת שגם דרכי הטבע המה נפלאות.

זַמְּרוּ לַה' חֲסִידָיו וְהוֹדוּ לְזֵכֶר קָדְשׁוֹ. (תהלים ל)

רש"י: זמרו לה' - על מה שעשה לי כי יכולים אתם לחסות בו שייטיב לכם ואפילו אתם שרוים בצער אל תיראו.

הָפַכְתָּ מִסְפְּדִי לְמָחוֹל לִי פִּתַּחְתָּ שַׂקִּי וַתְּאַזְּרֵנִי שִׂמְחָה: **לְמַעַן** יְזַמֶּרְךָ כָבוֹד וְלֹא יִדֹּם ה' אֱלֹהַי לְעוֹלָם אוֹדֶךָּ. (תהלים ל)

מלבי"ם: אחר שעיקר התכלית שבעבורו החיית אותי, היה כדי שאזמר לך, לכן גם אני אעשה כן ולעולם אודך:

כִּי לֹא שְׁאוֹל תּוֹדֶךָּ מָוֶת יְהַלְלֶךָּ לֹא יְשַׂבְּרוּ יוֹרְדֵי בוֹר אֶל אֲמִתֶּךָ: חַי חַי הוּא יוֹדֶךָ כָּמוֹנִי הַיּוֹם אָב לְבָנִים יוֹדִיעַ אֶל אֲמִתֶּךָ. (ישעיהו לח)

רד"ק: אחר שהוא רוצה וברא בני אדם אם כן הוא רוצה בקיומם ובלכתם בדרך טובה ושיהיה בהם דעת להבין בחסדי הבורא ולהודות לפניו עליהם כי זה מהדרך הטובה וכאשר לבם נכון הוא מסייע אותם ועוזרם בזה.

הוֹדוּ לַה' בְּכִנּוֹר בְּנֵבֶל עָשׂוֹר זַמְּרוּ לוֹ. (תהלים לג, ב)

שתי ידות: השיר והשבח להשי"ת ראוי שיהיה בתכלית השלמות לכן נאמר בנבל עשור כלומר בכל מאמצי כח.

אוֹדֶה ה' מְאֹד בְּפִי וּבְתוֹךְ רַבִּים אֲהַלְלֶנּוּ. (תהלים קט, ל)

פני יהושע (ברכות נג.) ברכת הודאה שייך בה יותר הטעם ברב עם הדרת מלך כדכתיב בתוך רבים אהללנו.

תהלים פרק סג (ו) כְּמוֹ חֵלֶב וָדֶשֶׁן תִּשְׂבַּע נַפְשִׁי וְשִׂפְתֵי רְנָנוֹת יְהַלֶּל פִּי:

מצודת דוד: כמו חלב ודשן - כמו שתשבע הנפש ממאכלי חלב ודשן כן תשבע נפשי אם יהלל פי את ה' בהרבה לשונות של רננות.

במגילת אסתר כתוב: בלילה ההוא נדדה שנת המלך ויאמר להביא את ספר הזכרונות דברי הימים ויהיו נקראים לפני המלך, וימצא כתוב אשר הגיד מרדכי וכ''ו, י''ל ק''ו מה אחשורוש הרשע כתב טובת מרדכי הצדיק בספר הזכרונות, ומזה היה תוקף הנס, כ''ש שיש לנו לכתוב טובות השי''ת אשר עושה לנו בספר זכרונות, ובודאי שעי''ז נזכה לישועות גדולים עד בלי די.

תהלים פרק יח מְהֻלָּל אֶקְרָא ייְיָ וּמִן אֹיְבַי אִוָּשֵׁעַ:

דרש דרש אמרו: שאדם יש לו צרה הוא אומר אוי וי, על כך אומר הפסוק יהודי היקר אם אתה רוצה להנצל מהצרות, יש לנו פתרון נפלא בשבילך: מהלל אקרא ה' – תהלל את ה' על החסדים הגדולים שעושה לך, ועל ידי כך מן אוי וי - מכל האוי וי תוושע.

לה' הישועה על עמך ברכתך סלה. להקדוש ברוך הוא לעשות נסים ונפלאות ולנו להללו ולברכו שנאמר על עמך ברכתך סלה. (מגילת תענית)

מָה אָשִׁיב לַיקֹוָק כָּל תַּגְמוּלוֹהִי עָלָי. (תהלים קטז, יב)

מלבי"ם: שעקר הטובה אצלו אינו החיים, רק עקר הטובה היא מה שעתה יוכל להודות לה'.

וז"ש מה אשיב לה' על מה שגמל עלי אשר כוס ישועות אשא ואקרא בשם ה' להודות לה' חסדו ולפרסם גדולת ה', ולשלם נדרי לה' נגדה נא לכל עמו, **שזה הוא הטובה היותר גדולה שגמל עלי**:

מִי הָאִישׁ הֶחָפֵץ חַיִּים (שחפץ שחייו יהיו חיים שמחים ומאושרים) אֹהֵב יָמִים (שאוהב את הימים העכשוויים) לִרְאוֹת טוֹב (ילמד לראות את הטוב שבכל יום) כיצד? יקח פנקס וירשום בכל יום את הדברים הנפלאים שעברו עליו.

וְזָכַרְתָּ אֶת יְדֹוָד אֱלֹקֶיךָ כִּי הוּא הַנֹּתֵן לְךָ כֹּחַ לַעֲשׂוֹת חָיִל (דברים ח,יח). כתב האור החיים: פירוש שצריך לתת לבו על טובתו כי מה' היתה לו.

כְּתֹב זֹאת זִכָּרוֹן בַּסֵּפֶר (שמות יז,יד) כתב האבן עזרא: ספר היה להם ובו היום כותבים את המלחמות שהיה עושה להם ה'.

אֲהַלְלָה שֵׁם אֱלֹהִים בְּשִׁיר וַאֲגַדְּלֶנּוּ בְתוֹדָה: וְתִיטַב לַיקֹוָק מִשּׁוֹר פָּר מַקְרִן מַפְרִיס: (תהלים פרק סט, לא-לב)

רד"ק: אהללה שם אלהים בשיר, ואגדלנו בתודה, שאודה חסדו ואקריב לפניו קרבן תודה. ותיטב לה'. התהלה שאהללנו בשיר תיטב לו משור פר.

מהר"ל: ראה הלשון היא גדולה מן הקרבנות שנאמר (תהילים סט) "אהללה שם אלהים בשיר ואגדלנו בתודה ותטב לה' משור פר". **ראה כמה גדול כח הדבור** שהרי על ידי הדבור הוא נותן הודאה אל השם יתברך **וההודאה היא על כל הקרבנות** וכמו שמבואר במקום אחר. וכל זה כי יותר נחשב הדבור שהוא שכלי מן הקרבנות עצמם שהם ע"י דברים גשמיים, ולפיכך כתיב "ותיטב לה' משור פר" כי הדבור הוא שכלי לכך הוא יותר נחשב מן הקרבנות, והבן זה מה שרמז כאן על הלשון שיש בו כח הודאה.

אֲבָרֲכָה אֶת ייְיָ בְּכָל עֵת תָּמִיד תְּהִלָּתוֹ בְּפִי: (תהלים לד, ב)

רד"ק: כי תמיד הוא עושה לי נסים וחייב אני לברכו ולהללו תמיד בכל עת.

כִּי שִׂמַּחְתַּנִי ייְיָ בְּפָעֳלֶךָ בְּמַעֲשֵׂי יָדֶיךָ אֲרַנֵּן: (תהלים צב ה)

בפעליך - במעשה הנסים שעשית לי: ונגד הטבע הסדורה בשלימות מראשית קדומים אמר במעשי ידיך ארנן:

וַיַּרְא אֱלֹקִים אֶת כָּל אֲשֶׁר עָשָׂה וְהִנֵּה טוֹב מְאֹד. (בראשית א,לא)

ה' עשה טוב, הודיענו שהעולם טוב, והוא מבקש מאתנו שנחקה אותו "והלכת בדרכיו" מה הוא אומר על עולמו טוב אף תפקידנו לומר ולהרגיש שהעולם טוב מאד. (ר' אביגדור מילר)

זֵכֶר רַב טוּבְךָ יַבִּיעוּ וְצִדְקָתְךָ יְרַנֵּנוּ: (תהלים קמה ז)

כאשר יזכרו את רוב טובו של ה' שהשפיע עליהם, אז "**יביעו**" תהא תודתם לה' נובעת מאליה, כמעין המתגבר.

דִּשַּׁנְתָּ בַשֶּׁמֶן רֹאשִׁי כּוֹסִי רְוָיָה (תהילים כג)

אנשים אומרים להסתכל על חצי הכוס המלאה אבל דוד המלך אומר "כוסי רויה" כוסי מלאה בטובך, אשר גמלת עמדי, עד אשר עולה היא על גדותיה בתודתי.

"טוב להודות לה" (תהילים צ"ב) איזוהי דרך טובה שיבור לו האדם? זוהי ההודיה. אם מבקש אדם לדעת מהו הטוב הנדרש ממנו, אומר לו דוד המלך: "טוב להודות לה'". זהו מה שאלוקיך שואל מעמך, כי לכך נוצרת.

מְהֻלָּל אֶקְרָא ייְיָ וּמִן אֹיְבַי אִוָּשֵׁעַ: תהלים פרק יח

דרך בני אדם בשעת כאב לזעוק "אוי" "וי" ועל כך אומר הפסוק: הזוכה להלל את ה' נושע מכל האוי-וי.

### פסוקים בארוכה.

תהלים פרק ו

כִּי אֵין בַּמָּוֶת זִכְרֶךָ בִּשְׁאוֹל מִי יוֹדֶה לָּךְ:

צ''ע האם האדם שמת אינו מודה להקב''ה?

רד"ק: אם אמות לא אזכרך במותי ולא אודה לך, ואם תרפאני אודה לך לעיני כל. כמו שאמר חזקיהו (ישעיה לח, יח): כי לא שְׁאוֹל תּוֹדֶךָּ מות יהללך וגו', ואמר דוד (מזמור קיח, יז): לא אמות כי אחיה ואספר מעשי יה. כי **הגוף** אחר צאת הנשמה כאבן דומם, והוא יורד שאולה, אבל הנשמה תעלה ותודה ותשבח לעד בלא הפסק. אבל **יתאוה הצדיק לחיות לעשות רצון האל בעודנו חי** להרבות שכר הנשמה לעולם הבא:

מלבי"ם ביאור המילות: מות, שאול. השאול הוא הקבר. שהוא אחר המות, ויצייר שבמות עדיין יש הודאה, כי עד שם יגיעו טובותיו באיכות המות שיבחר לו מיתה קלה ויודה על טובותיו כל ימי חייו, רק התהלה וזכר גבורותיו נפסקו אז, ובשאול נפסק גם ההודאה וכה"א כי לא שאול תודך מות יהללך (ישעי' ל"ח):

מלבי"ם: וגם הלא בשאול מי יודה לך, כי הנעדר לא יקבל עוד טובך וחסדך, אשר זה תכלית הבריאה אשר בראת להיטיב להנמצאים (כדי) שהם יודוך:
