שירים חסידיים

תְּהֵא הַשָׁעָה הַזֹאת שְׁעַת רַחֲמִים וְעֵת רָצוֹן מִלְפָנֶיךָ:

\* \* \*

אַשְׁרֵי מִי שֶׁעֲמָלוֹ בָּתּוֹרָה וְעוֹשֶׂה נַחַת רוּחַ לְיוֹצְרוֹ:

\* \* \*

כִּי הֵם חַיֵינוּ וְאוֹרֵךְ יָמֵינוּ וּבָהֶם נֶהְגֶה יוֹמָם וַלָיְלָה:

\* \* \*

עוֹז וְהָדָר לְבוּשָׁה וַתִּשְׂחָק לְיוֹם אַחַרוֹן:

\* \* \*

טוֹב לְהוֹדוֹת לה' וּלְזַמֵר לְשִׁמְךָ עֶלְיוֹן:

\* \* \*

טוֹב לִי תּוֹרַת פִּיךָ מֵאַלְפֵי זָהָב וָכֶסֶף:

\* \* \*

וְתֵן בָּנוּ יֵצֵר טוֹב לְעָבְדְךָ בֶּאֱמֶת:

\* \* \*

יְהִי הַחוֹדֶשׁ הַזֶּה כִּנְבוּאַת אֲבִי חוֹזֶה וְיִשָׁמַע בְּבַיִת זֶה קוֹל שָׂשׂוֹן וְקוֹל שִׂמְחָה, חָזָק יְמַלֵא מִשְׁאֲלוֹתֵינוּ אֲמִיץ יַעֲשֶַׂה בַּקָשׁוֹתֵינוּ וְהוּא יִשְׁלַח בְּמַעֲשֶׂה יָדֵינוּ בְּרָכָה וְהַצְלָחַה:

\* \* \*

יִבָּנֶה הַמִקְדָשׁ עִיר צִיוֹן תְּמַלֵא, וְשָׁם נָשִׁיר שִׁיר חָדָשׁ וּבִּרְנָנָה נַעֲלֶה:

\* \* \*

וּבְנֵה יְרוּשָׁלַיִם עִיר הַקוֹדֶשׁ בִּמְהֵרָה בְּיָמֵינוּ:

\* \* \*

עוּצוּ עֵצָה וְתּוּפָר דַבְּרוּ דָבָר וְלֹא יָקוּם כִּי עִמָּנוּ אֵל:

\* \* \*

עִבְדוּ אֶת ה' בְּשִׂמְחָה בּוֹאוּ לְפָנָיו בִּרְנָנָה:

\* \* \*

מְקִימִי מֵעָפָר דָל:

\* \* \*

סִימָן טוֹב וּמַזָל טוֹב יְהֵא לָנוּ וּלְכָל יִשְׂרָאֵל:

**\* \* \***

וְטַהֵר לִבֵּנוּ לְעָבְדְךָ בֶּאֱמֶת:

\* \* \*

מִצְוָה גְּדוֹלָה לִהְיוֹת בְּשִׂמְחָה תָמִיד:

\* \* \*

כִּי אַתָּה הוּא מֶלֶךְ מַלְכֵי הַמְלָכִים מַלְכוּתוֹ נֶצַח נוֹרְאוֹתָיו שִׂיחוּ סַפְּרוּ עֻזוֹ פֵאֲרוּהוּ צְבָאָיו, קַדְשׁוּהוּ רוֹמְמוּהוּ רוֹן שִׁיר וָשֶׁבַח תּוֹקֵף תְּהִילַת תִּפְאֲרְתוֹ:

\* \* \*

אֲנַחְנוּ מַאֲמִינִים בְּנֵי מַאֲמִינִים וְאֵין לָנוּ עַל מִי לְהִישָׁעֵן אֶלָא עַל אָבִינוּ שֶׁבַּשָׁמַיִם:

\* \* \*

אָבִינוּ מַלְכֵּנוּ פְתַח שַׁעֲרֵי שָׁמַיִם לִתְפִילָתֵנוּ:

\* \* \*

שַׂמְחֵם בְּבִנְיָן שָׁלֵם:

\* \* \*

תּוֹרַת ה' תְּמִימָה מְּשִׁיבַת נָפֶשׁ:

\* \* \*

צִיוֹן הֲלֹא תִּשְׁאֲלִי לִשְׁלוֹם אֲסִירַיִךְ:

\* \* \*

אָבִינוּ אָב הַרַחֲמָן הַמְּרַחֵם עָלֵינוּ וְתֵן בְּלִבֵּנוּ בִּינָה לְּהָבִין וּלְהַשְׂכִּיל לִשְׁמוֹע לִלְמוֹד וּלְלַמֵד לִשְׂמוֹר וְלַעֲשׂוֹת וּלְקַיֵים אֶת כָּל דִבְרֵי תַלְמוּד תּוֹרָתֶךָ בְּאַהֲבָה, וְהָאֵר עֵינֵינוּ בְּתוֹרָתֶךָ וְדַבֵּק לִבֵּנוּ בְּמִצְוֹותֶיךָ וְיַחֵד לְבָבֵנוּ לְאַהֲבָה וּלְיִרְאָה אֶת שְׁמֶךָ, לֹא נֵבוֹשׁ וְלֹא נִכָּלֵם וְלֹא נִכָּשֵׁל לְעוֹלָם וָעֶד:

\* \* \*

אַשְׁרֵי יוֹשְׁבֵי בֵיתֶךָ עוֹד יְהַלְלוּךָ סֶלָה:

\* \* \*

יְבָרֶכְךָ ה' מִצִיוֹן וּרְאֵה בְּטוּב יְרוּשָׁלַיִם כָּל יְמֵי חַיֶיךָ וּרְאֵה בָנִים לְבָנֶיךָ שָׁלוֹם עַל יִשְׂרָאֵל:

\* \* \*

אֲנִי מַאֲמִין בֶּאֱמוּנָה שְּלֵמָה בְּבִיאַת הַמָשִׁיחַ וְאַף עַל פִּי שֶׁיִתְמַהְמֵהַ עִם כָּל זֶה אֲחַכֶּה לוֹ בְּכָל יוֹם שֶׁיָבוֹא:

\* \* \*

קוֹל בְּרָמָה נִשְמַע נֶהִי בֶכִי תַמְרוּרִים רָחֵל מְבַכַּה עַל בָּנֶיהָ מֵאֲנָה לְהִנָחֵם, כֹּה אָמַר ה' מִנְעִי קוֹלֵךְ מִבְּכִי וְעֵינַיִךְ מִדִמְעָה כִּי יֵשׁ שָׂכָר לִפְעוּלָתֵךְ וְשָׁבוּ בָּנִים לִגְבוּלָם וְיֵשׁ תִּקְוָה לְאַחֲרִיתֵךְ וְשָׁבוּ מֵאֶרֶץ אוֹיֵב:

\* \* \*

אוֹהֲבֵי ה' שִׂנְאוּ רָע שׁוֹמֵר נַפְשׁוֹת חֲסִידָיו מִיַד רְשָׁעִים יַצִילֵם:

\* \* \*

אוֹר זָרוּעַ לַצַדִיק וּלְיִשְׁרֵי לֵב שִׂמְחָה:

\* \* \*
