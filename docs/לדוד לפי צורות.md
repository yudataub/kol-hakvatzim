יְהוָה מָעוֹז חַיַּי מִמִּי אֶפְחָד:- אִם תָּקוּם עָלַי מִלְחָמָה בְּזֹאת אֲנִי בוֹטֵחַ:

{ב} בִּקְרֹב עָלַי מְרֵעִים לֶאֱכֹל אֶת בְּשָׂרִי צָרַי וְאֹיְבַי לִי הֵמָּה כָשְׁלוּ וְנָפָלוּ: {ג {ד} אַחַת שָׁאַלְתִּי מֵאֵת יְהוָה אוֹתָהּ אֲבַקֵּשׁ שִׁבְתִּי בְּבֵית יְהוָה כָּל יְמֵי חַיַּי לַחֲזוֹת בְּנֹעַם יְהוָה וּלְבַקֵּר בְּהֵיכָלוֹ: {ה} כִּי יִצְפְּנֵנִי בְּסֻכֹּה בְּיוֹם רָעָה יַסְתִּרֵנִי בְּסֵתֶר אָהֳלוֹ בְּצוּר יְרוֹמְמֵנִי: {ו} וְעַתָּה יָרוּם רֹאשִׁי עַל אֹיְבַי סְבִיבוֹתַי וְאֶזְבְּחָה בְאָהֳלוֹ זִבְחֵי תְרוּעָה אָשִׁירָה וַאֲזַמְּרָה לַיהוָה: {ז} שְׁמַע יְהוָה קוֹלִי אֶקְרָא וְחָנֵּנִי וַעֲנֵנִי: {ח} לְךָ אָמַר לִבִּי בַּקְּשׁוּ פָנָי אֶת פָּנֶיךָ יְהוָה אֲבַקֵּשׁ: {ט} אַל תַּסְתֵּר פָּנֶיךָ מִמֶּנִּי אַל תַּט בְּאַף עַבְדֶּךָ עֶזְרָתִי הָיִיתָ אַל תִּטְּשֵׁנִי וְאַל תַּעַזְבֵנִי אֱלֹהֵי יִשְׁעִי: {י} כִּי אָבִי וְאִמִּי עֲזָבוּנִי וַיהוָה יַאַסְפֵנִי: {יא} הוֹרֵנִי יְהוָה דַּרְכֶּךָ וּנְחֵנִי בְּאֹרַח מִישׁוֹר לְמַעַן שׁוֹרְרָי: {יב} אַל תִּתְּנֵנִי בְּנֶפֶשׁ צָרָי כִּי קָמוּ בִי עֵדֵי שֶׁקֶר וִיפֵחַ חָמָס: {יג} לוּלֵא הֶאֱמַנְתִּי לִרְאוֹת בְּטוּב יְהוָה בְּאֶרֶץ חַיִּים: {יד} קַוֵּה אֶל יְהוָה חֲזַק וְיַאֲמֵץ לִבֶּךָ וְקַוֵּה אֶל יְהוָה:

תהלים פרק כז

לְדָוִד יְקֹוָק אוֹרִי וְיִשְׁעִי מִמִּי אִירָא יְקֹוָק מָעוֹז חַיַּי מִמִּי אֶפְחָד: בִּקְרֹב עָלַי מְרֵעִים לֶאֱכֹל אֶת בְּשָׂרִי צָרַי וְאֹיְבַי לִי הֵמָּה כָשְׁלוּ וְנָפָלוּ: אִם תַּחֲנֶה עָלַי מַחֲנֶה לֹא יִירָא לִבִּי אִם תָּקוּם עָלַי מִלְחָמָה בְּזֹאת אֲנִי בוֹטֵחַ: אַחַת שָׁאַלְתִּי מֵאֵת יְקֹוָק אוֹתָהּ אֲבַקֵּשׁ שִׁבְתִּי בְּבֵית יְקֹוָק כָּל יְמֵי חַיַּי לַחֲזוֹת בְּנֹעַם יְקֹוָק וּלְבַקֵּר בְּהֵיכָלוֹ: כִּי יִצְפְּנֵנִי בְּסֻכֹּה בְּיוֹם רָעָה יַסְתִּרֵנִי בְּסֵתֶר אָהֳלוֹ בְּצוּר יְרוֹמְמֵנִי: וְעַתָּה יָרוּם רֹאשִׁי עַל אֹיְבַי סְבִיבוֹתַי וְאֶזְבְּחָה בְאָהֳלוֹ זִבְחֵי תְרוּעָה אָשִׁירָה וַאֲזַמְּרָה לַיקֹוָק: שְׁמַע יְקֹוָק קוֹלִי אֶקְרָא וְחָנֵּנִי וַעֲנֵנִי: לְךָ אָמַר לִבִּי בַּקְּשׁוּ פָנָי אֶת פָּנֶיךָ יְקֹוָק אֲבַקֵּשׁ: אַל תַּסְתֵּר פָּנֶיךָ מִמֶּנִּי אַל תַּט בְּאַף עַבְדֶּךָ עֶזְרָתִי הָיִיתָ אַל תִּטְּשֵׁנִי וְאַל תַּעַזְבֵנִי אֱלֹהֵי יִשְׁעִי: כִּי אָבִי וְאִמִּי עֲזָבוּנִי וַיקֹוָק יַאַסְפֵנִי: הוֹרֵנִי יְקֹוָק דַּרְכֶּךָ וּנְחֵנִי בְּאֹרַח מִישׁוֹר לְמַעַן שׁוֹרְרָי: אַל תִּתְּנֵנִי בְּנֶפֶשׁ צָרָי כִּי קָמוּ בִי עֵדֵי שֶׁקֶר וִיפֵחַ חָמָס: לוּלֵא הֶאֱמַנְתִּי לִרְאוֹת בְּטוּב יְקֹוָק בְּאֶרֶץ חַיִּים: קַוֵּה אֶל יְקֹוָק חֲזַק וְיַאֲמֵץ לִבֶּךָ וְקַוֵּה אֶל יְקֹוָק:

ה' אורי – הצרה משולה לחושך והישועה לאור.

לְדָוִד יְקֹוָק אוֹרִי וְיִשְׁעִי מִמִּי אִירָא יְקֹוָק מָעוֹז חַיַּי מִמִּי אֶפְחָד:

מדוע שה' אורי וישעי – לא אירע ואפחד?

יש מסבירים לפי הרמב"ם שמי שדבוק בהשגחתו יתברך לא יאונה לו כל רע.

עוד אפשר כאשר אדם מרגיש שה' אורי וישעי הרי אין לו פחד משום דבר – משום שהוא בטוח שמה שה' יביא זה הכי טוב בשבילו.

אִם תָּקוּם עָלַי מִלְחָמָה בְּזֹאת אֲנִי בוֹטֵחַ:

מהו בזאת?

לפני או אחרי.

מה בין שאלה לבקשה?

שאלה – אדם מבקש כסף. אבל הוא רוצה לקנות כמה וכמה דברים.

בקשה – זה מה המבוקש מהדבר.

מדוע הוא רוצה רק לבקר בהיכלו?

יש מעלה בישיבה הקבועה ויש מעלה בביקור – שאז מתרגשים.

רש"י מפרש: לבקר – לבוא כל בוקר ובוקר.

ואפשר לבקר בהיכלו – הכוונה להרגיש את מלכותו.

כיצד אומר דוד שהוא לא פחד 'ממי אירא', הרי כתוב מפורש תהלים פרק יח פסוק ה - אֲפָפוּנִי חֶבְלֵי מָוֶת וְנַחֲלֵי בְלִיַּעַל יְבַעֲתוּנִי:.

יש מצב של גדלות המוחין ויש מצב של קטנות המוחין.

בחלק הראשון של המזמור מדבר על השאיפה והרצון

בחלק השני מבקש שיזכה לזה.

וּנְחֵנִי בְּאֹרַח מִישׁוֹר לְמַעַן שׁוֹרְרָי:

למען שוררי – למען שלא ישמחו אויבי המביטים בי לרעה.

מה בין דרך לאורח?

קוה אל ה' פעמים

ללמד שיחזור ויתפלל.

ויש אומרים שזה סגולת חזרה הש"צ.

הקווי לזולת מחליש את הלב

הקווי לה' מחזק את הלב.

ועוד שהקווי בעצם הוא המטרה – ולכן הוא חזק ויאמץ לבך.
