נוסח הודאה לה'

(מהרה''ק ר' אהרן ראטה זיע''א)

רִבּוֹן הָעוֹלָמִים אַנְתְּ הוּא עִלַּת הָעִלּוֹת וְסִיבַּת כָּל הַסִיבּוֹת שַׁלִיט בָּעֶלְיוֹנִים וּבַתַחְתּוֹנִים אַנְתְּ רִבּוֹן תַקִיף וּבַעַל הַיְכוֹלֶת וּבַעַל הַכּוֹחוֹת כּוּלָם, מִי יָכוֹל לְשַׁעֵר וּלְסַפֵּר אַף אֶפֶס קְצָת מִגְדוּלָתְּךָ וּמִגְּבוּרָתְּךָ, וְאַתָּה אֲצָלְתָּ וּבָרָאתָּ וְיָצַרְתָּ וְעָשִׂיתָּ כָּל הָעוֹלָמוֹת וְכָל הַבְּרוּאִים וְהַיְצוּרִים וְהַנַעֲשִׂים וְהַכֹּל כְּאֶפֶס וּכְאַיִן נֶגְדֶךָ, וְאַתָּה מְחַדֵשׁ בְּטוּבְךָ בְּכָל יוֹם תָמִיד מַעֲשֶׂה בְּרֵאשִׁית, וְאַתָּה מְחַיֶה וּמְהַוֶה אֶת כּוּלָם בְכָל עֵת וּבְכָל שָׁעָה וּבְכָל רֶגַע, וְאִלְמָלֵא יְצוּיָיר הֶעְדֵר שִׁפְעֲךָ עַל רֶגַע חָלִילָה הָיוּ הַכֹּל בְּטֵלִין וּמְבוּטָלִין, וְאַתָּה בְּטוּבְךָ הַגָדוֹל מְרַחֵם עַל כָּל הַבְּרוּאִים שֶׁבָּרָאתָ בְּעוֹלָמֶךָ וְאַתָּה זָן וּמְפַרְנֵס אֶת כּוּלָם מִקַרְנֵי רְאֵמִים וְעַד בֵיצֵי כִּנִים, וְאֵין כָּל דָבָר נֶעְלָם מִמְּךָ וְאֵין נִסְתָּר מִנֶגֶד עֵינֵיךָ כִּי הַכֹּל מוּשְׁגַח מֵאִתְּךָ בְּהַשְׁגָחָה פְרָטִית.

אָבִינוּ אָב הַרַחֲמָן וּבַעַל הַחֶסֶד מוֹדִים אֲנַחְנוּ לְפָנֶיךָ עַל הַחֲסָדִים הַגְּדוֹלִים וְהַמְּרוּבִּים שֶׁעָשִׂיתָּ וְשֶׁאַתָּה עוֹשֶׁה, וְעַל הַחֲסָדִים שֶׁאַתָּה עָתִיד לַעֲשׂוֹת עִמָּנוּ, וּבִפְרָט עַל הַחֲסָדִים הַגְּדוֹלִים וְהַעֲצוּמִים שֶׁאַתָּה זָן וּמְפַרְנֵס וּמְחַיֶה אוֹתָנוּ בְּכָל עֵת וּבְכָל שָׁעָה בְּנִפְלְאוֹתֶיךָ הַמְּרוּבִּים, וּבְכָל מִינֵי חַסֲדִים וְהַשְׁפָּעוֹת טוֹבוֹת שֶׁאֵין לָהֶם שִׁיעוּר וָחֵקֶר שֶׁאַתָּה עוֹשֶׂה עִמָּנוּ תָּמִיד, וְעַל הַכֹּל אֲנַחְנוּ מוֹדִים לְךָ וּמְבָרְכִים וּמְשַׁבְּחִים לְשִׁמְךָ הַגָּדוֹל וְהַקָדוֹשׁ עַל כָּל נְשִׁימָה וּנְּשִׁימָה, כִּי לֹא לְפִי מַעֲשֵׂינוּ אַתָּה עוֹשֶׁה עִמָּנוּ כִּי אֵין אֲנַחְנוּ רְאוּיִם כְלוּם לְהַחֲסָדִים הַגְּדוֹלִים שֶׁאַתָּה עוֹשֶׂה עִמָּנוּ בְּכָל עֵת, כִּי הֵרַעְנוּ מַעֲשֵׂינוּ וְהִכְעַסְנוּ אוֹתְךָ בְּמֲעַשֵׂינוּ וּבְמִדוֹתֵינוּ הַרָעִים חָטָּאנוּ וְעָוִינוּ וּפָשַׁעְנוּ וּפָגַמְנוּ בְמַחֲשָׁבָה דִיבּוּר וּמַעֲשֶה בְּכָל מִינֵי חֲטָאִים וְעֲוֹנוֹת וּפְשָׁעִים, וְהֵמָּה כְּהָרִים וּגְבָעוֹת וְרַבּוּ וְעָצְמוּ בְּלִי שִׁיעוּר וּבְלִי מִסְפָּר. פָּגַמְנוּ אֶת רמ''ח אֵבָרֵינוּ ושס''ה גִידֵינוּ עַד שֶׁאֵין בָּנוּ מְתוֹם.

רִבּוֹנוֹ שֶׁל עוֹלָם בַּמֶה אֶתְרַצֶה וְאֶתְפַּיֵיס לְפָנֶיךָ, וְאַף אִם אֶתֵן לְךָ כָּל חֲלָבִי וְדָמִי וְאֶמְסוֹר לְךָ כָּל חָיַי, מַה כָּל חָיַי, הֶבֶל וָרִיק, רַק בָּזֶה בָטַחְתִּי בְּגוֹדֶל חֲסָדֶיךָ וְרַחֲמֶיךָ וְגוֹדֶל עַנְוְתָנוּתֶיךָ כִּי בִמְקוֹם גְּדוּלָתֶךָ שָׁם עַנְוְתָנוּתֶיךָ, וְאַף כִּי גְלוּיִים לְךָ כָּל הַנִסְתָּרוֹת וְהֶעָבָר וְהַהֹוֶה וְהֶעָתִיד לָבֹא שָׁוִים בְּעֵינֶיךָ, וְאַף קוֹדֶם שֶׁבְרָאַתָנִי הָיוּ גְלוּיִים לְפָנֶיךָ כָּל עִנְינַי וְחֲטָאַי וְעֲוֹנֹתַי וּפְּשָׁעַי, אַף עַל פִּי כֵן לֹא מָנַעְתָּ חֲסְדְךָ מִמֶנִי וּבָרָאתָּ אוֹתִי מִטִיפָּה סְרוּחָה, וְנִסִים וַחֲסָדִים הִפְלֵאתָ עִמִּי בְּעֵת הִתְרַקְמִי וְהִתְצַיְירִי, וְכֹחֲךָ הִשְׁפַעְתָּ לְשִׁבְעַת כוֹכְבֵי לֶכֶת וי''ב מַזָלוֹת הָרָקִיעַ הַמְּמוּנִים עַל צִיוּר וְרִיקוּם הַוָלָד, כֹּל אֶחָד עוֹשֶׁה אוּמָנוּתוֹ.

וְאַחַר צִיוּר הָעִיבּוּר וְתִיקוּן אֵבָרַי וְהִתְקַשֵׁר בִּי נֶפֶשׁ הַחַי, עָמַדְתִּי בִּמְעֵי אִמִּי מְקוּפָּל וּמוּנָח כְּפִנְקָס וְהוּא מָקוֹם צַר, שְׁתֵּי יָדַי עַל שְׁתֵּי צְדָעַי, וּשְׁתֵּי אֲצִילַי עַל שְׁתֵּי אֲרְכּוּבוֹתַי, וּשְׁתֵּי עֲקֵבַי עַל שְׁתֵּי עַגְבוֹתַי, וְרֹאשִֹי מוּנָח בֵּין בִּרְכַּי וּפִי סָתוּם וְטַבּוּרִי פָּתוּחַ, וְלֹא יָצָא מִמֶנִּי הָרוּחַ, וּמִדָם הֲנִּדוּת הָיָה מְזוֹנִי, וְלֹא הוֹצֵאתִי הָרֵעִי מִבִטְנִי שֶׁלֹא לַהֲרוֹג אֶת אִמִי, וְאַחַר כָּך יָצָאתִי מִבֶטֶן אִמִי בְדֶרֶךְ צַר מִבְּלִי תַחְבּוּלָה שֶׁאֶתְחַכֵּם בָּה עַל צֵאתִי וְלֹא דָבָר הַמְסַיֵיעַ לִי בּוֹ אֶלָא בִּיְכוֹלֶת הַרַחוּם וְחַנוּן, וְאַחַר צֵאתִי מֵרֶחֶם אִמִי הִזְמִין הַשֵׁם חָלָב בִשְׁדֵי אִמִי נִגָר עָרֵב וּמָתוֹק, וְנָתַן אוֹתִי לְחֵן בְּעֵינֵי יוֹלְדִי, וְהֵקֵל בְּעֵינֵיהֶם כָּל טָרְחִי וְעָמָל גִדוּלִי וְלִרְחוֹץ צוֹאָתִי, וְיָצָאתִי מִמְעֵי אִמִי מָלֵא רִירִין וּמָלֵא דָם וְלֹא נִמְאַסְתִּי וְלֹא נִתְרַחֲקְתִּי מִפְנֵי זֶה מֵאֲבוֹתַי אַדְרַבָּה הֵמָה נִשְׁקוּ וְחִבְבוּ אוֹתִי.

וַיְכוֹנֵנִי הַבּוֹרֵא בְּחָכְמָה בִּינָה וַדַעַת וּבָרָא בִּי נְקָבִים נְקָבִים, וּמִחוּץ עָשָׁה לִי נִקְבֵי עֵינַיִם לְהַבִּיט וְלִרְאוֹת בָּהֶם, עָשָׁה לִי עֲפְעַפַּיִם לְהַעֲצִים עֵינַי וְלִישַׁן וְשֶׁלֹא אַבִּיט בְּרָע, עָשָׂה לִי נִקְבֵי הָאָזְנַיִם לִשְׁמוֹעַ דְבָרִים שֶׁבַקְדוּשָׁה, וְהַאָלְיָה לָכוֹף בְּתוֹכָם שֶׁלֹא לְקַבֵּל שְׁמוּעוֹת הַאֲסוּרוֹת, עָשָׂה לִי נִקְבֵי הַחוֹטֶם לְהָרִיחַ וּלְהוֹרִיד בָּה הָעֲדָפַת לֵיחוּת הָרֹאשׁ, עָשָׂה לִי הַפֶּה לְדַבֵּר דְבָרִים קְדוֹשִׁים וְלִלְמוֹד תּוֹרָה וּלְהִתְפַּלֵל וְלֶאֶכוֹל וְלִשְׁתּוֹת, עָשָׂה לִי שִׁינַיִם לִטְחוֹן הַמַאֲכָל, עָשָׂה לִי הַלָשׁוֹן לָלוּשׁ הַמַאֲכָל בְּתּוֹך הַפֶּה וּכְדֵי לְחַתֵּךְ הַדִיבּוּר, עָשָׂה לִי בֵּית הַבְּלִיעָה וְהַוֵשֶט לִבְלוֹעַ הַמַאֲכָל וְהַמַשְׁקֶה, עָשָׂה לִי גַרְגֶרֶת לְהַעֲלוֹת וּלְהוֹרִיד נִשְׁמַת רוּחַ חַיִים מִן הַלֵב מִתּוֹךְ הָרֵיאָה אֶת הַנְּחִירַיִם וְאֶל הַפֶּה וּלְהוֹצִיא הֲבָרַת קוֹל הַדִיבּוּר הַיוֹצֵא מִתּוֹךְ סִמְפּוֹנֵי הָרֵיאָה הַבָּאָה מִנִשְׁמַת רוּחַ חַיִים.

עָשָׂה לִי הַלֵב לִהְיוֹת בֵּית מָכוֹן לְרוּחַ חַיִים וְלַחְשׁוֹב מַחֲשָׁבוֹת טוֹבוֹת, עָשָׂה לִי הַקֶרֶב הָעֶלְיוֹן הוּא הָאִצְטוֹמְכָא לְקַבֵּל הַמַאֲכָל וְהַמַשְקֶה בְּתוֹכוֹ וּלְכַלְכֵּל מִטַעַם הַמַאֲכָל וְהַמַשְׁקֶה כָּל הַגוּף לְהַחֲיוֹתוֹ, עָשָׂה לִי הַכָּבֵד לְצַד הַיָמִין עִם מָרָה הַאֲדוּמָה וְהַדָּם בִּמְקוֹרוֹת הַכָּבֵד לְחַמֵם הַקֶרֶב הָעֶלְיוֹן וּמֵחוֹם הַכָּבֵד וְהַמָרָה הַאֲדוּמָה וְהַדָּם מִתְבַּשֵׁל הַמַאֲכָל וְהַמַשְׁקֶה בְּתוֹךְ הַקֶרֶב הָעֶלְיוֹן, עָשָׂה לִי מֵעַיִים לְקַבֵּל פְסוֹלֶת הַמַאֲכָל וְהַמַשְׁקֶה וּמִתְגַלְגֵל בְּתּוֹכָם בְּעַקְמוּמִית לְאַט לְאַט עַד שֶּמַגִיעַ לִמְקוֹם הוֹצָאַת גְדוֹלִים וּקְטַנִים, עָשָׂה לִי שְׁתֵּי הַכְּלָיוֹת יָמִין וּשְׁמֹאל לְחַמֵּם הַבֶּטֶן הַתַחְתּוֹן בְּחוּמָם לַחוּתוֹ לְחַזְּקוֹ וּלְהוֹצִיא צְרָכָיו חוּץ לַגוּף כְּדֶרֶךְ הָאָדָם, וְעוֹד אָמְרוּ רַבּוֹתֵינוּ הַלֵב מֵבִין וְהַכְּלָיוֹת יוֹעֲצוֹת, עָשָׂה לִי הַטְחוֹל בְּצַד הַשְׂמֹאל עִם הַמָרָה הַשְׁחוֹרָה לְקָרֵר בְּקַרוּתָם הַקְרָבָיִים וְהַמֵעַיִים וְהַבֶּטֶן שֶׁלֹא יִתְיַבְשׁוּ וְיֵחָרְבוּ הַמַאֲכָל וְהַמַשְׁקֶה מִפְּנֵי חוֹם רִיתּוּחַ הַקְרָבָיִים הַמִּתְחַמְמִים וּמְרוּתָּחִים מֵחוֹם הַדָם וְהַכָבֵד וְהַמָרָה הַאֲדוּמָה, וְעוֹד אָמְרוּ רַבּוֹתֵינוּ טְחוֹל שׁוֹחֵק.

עָשָׂה לִי עָצָמוֹת הַשִׁדְרָה וּשְׁאָר עָצָמוֹת לִהְיוֹת בִּנְיַן גוּפִי חָזָק, שָׂם בָּשָׂר עַל הַעָצָמוֹת כְּדֵי לְחַמֵם קוֹר הַעָצָמוֹת, עָשָׂה לִי גִידֵי הַדָם הֵם הַוְורִידִין לְהַשְׁקוֹת אֶת הַגוּף וְהוֹלְכִים מִצַד לְצַד וּמִפֶּה לְפֶּה, עָשָׂה לִי מֵיתָרִים שֶׁמְחַזְקִים דֶבֶק הַעָצָמוֹת וְהָאֵבָרִים, עָשָׂה לִי אֵבָרִים וּקְשָׁרִים וּדְבָקִים וְחוּלְיוֹת עַל שִׁדְרָה וְקִשְׁרֵי אֶצְבָּעוֹת יָדַיִם וְרַגְלַיִם, וּזְרוֹעוֹת וִיְרֵכַיִם וּבִרְכַּיִם וְשׁוֹקַיִם וְקַרְסוּלַיִם וְכַּפּוֹת הָרַגְלַיִם וְחוּלְיוֹת הַצָוַאר וְשִׁכְמֵי הַכְּתֵפַיִם, כְּדֵי שֶׁאוּכַל לְהִתְפַּשֵׁט וְלִכְרוֹעַ וְלִכְפוֹף וְלִזְקוֹף וְלַעֲמוֹד וְלֵילֵךְ וְלֵישֵׁב.

**קָרַם** לִי עוֹר מִלְמַעְלָה לְבָּשָׂר וְלַעֲצָמוֹת וּלְגִידִים וּלְמֵיתָרִים כְּדֵי לְכַסוֹת אֶת הַגוּף לִהְיוֹת חַם מִלְהִתְקָרֵר וּלְהָגִין עַל לֵיחוּת הַגוּף וְהָאֵבָרִים הַפְּנִימִיִים, עָשָׂה לִי גוּלְגוֹלֶת הָרֹאשׁ לִהְיוֹת מֶלֶךְ עַל כֹּל הָאֵבָרִים וּמוֹחַ הָרֹאשׁ שֶׁשָׁם יִשְׁכּוֹן יִשׁוּב הַדַעַת וְהַנְּשָׁמָה וְהָחַכְמָה, עָשָׂה לִי לֵיחָה מִסָבִיב לְמוֹחַ הָרֹאשׁ כְּדֵי שֶׁלֹא יִתְיַבְּשׁוּ הַמוֹחַ וְהָעֵינַיִם מִחוֹם הַקְרָבָיִים הָעוֹלֶה מִלְמַטָה לְמַעְלָה אֶל מְקוֹם הָרֹאשׁ, עָשָׂה לִי שְׂעָרוֹת הָרֹאשׁ לְחַמֵם הָרֹאשׁ שֶׁלֹא יִתְקָרֵר עַל כִּי אֵין לָאָדָם בָּשָׂר בֵין הָעוֹר וּבֵין עֶצֶם הַגוּלְגוֹלֶת, עָשָׂה לִי שְׂעָרוֹת הַזָקָן מִפְּנֵי תוֹאַר הַדְרַת פָּנַי וְלִהְיוֹת מוּבְדַל הָאִישׁ בְּמַרְאִיתוֹ מִן הַנָשִׁים, עָשָׂה לִי טַבּוּר כְּדֵי לְקַבֵּל מִחְיָה בְּעוֹדִי בְּבֶטֶן אִמִי עַד צֵאתִי לַאֲוִיר הָעוֹלָם כִּי לֹא יָכוֹל הַוָלָד לְקַבֵּל מָזוֹן וּמִחְיָה מִן הַפֶּה עַל שֶׁהוּא סָתוּם עַד צֵאתוֹ לַאֲוִיר הָעוֹלָם, עָשָׂה לִי יָדַיִם לְמַשְׁמֵשׁ וְלִתְפוֹס וְלֶאֱחוֹז וּלְהַחֲזִיק וְלַעֲשׂוֹת מְלָאכָה כְּדֵי לְהַחֲיוֹת אֶת נַפְשִׁי וְלַעֲסוֹק בָּהֶם בְּמִצְוֹת וּמַעֲשִׂים טוֹבִים, עָשָׂה לִי רַגְלַיִם לְהַעֲמִיד עֲלֵיהֶם הַגוּף וּלְהוֹלִיכוֹ וְלַהֲבִיאוֹ בְּכָל מָקוֹם וְלֵילֵךְ וְלִרְדוֹף בָּהֶם אַחַר מִצְוֹתָיו יִתְבָּרַךְ.

וְאַחַר צֵאתִי לָאֲוִיר הָעוֹלָם גִדְּלַנִי הַבּוֹרֵא וַעֲזָרַנִי בְּכֹל מִינֵי עֵזֶר, וְכַמָּה מֵהַחֲסָדִים הַגְּדוֹלִים שֶׁעָשָׂה לִי הַבּוֹרֵא שֶׁבְּרָאָנִי בְּאֵבָרִים שְׁלֵמִים שֶׁאִלְמָלֵי הָיָה אֵיזֶה חִסָרוֹן לְהָאָדָם ח''ו בְּאֶחָד מֵאֵבָרָיו כַּמָה הָיָה מִשְׁתַּדֵל וּמְחַפֵּשׂ רוֹפֵא שֶׁיוּכַל לְרַפֵּא אוֹתוֹ, וְאִם אַחַר רוֹב הִשְׁתַדְלוּת הָיָה מוֹצֵא לוֹ אֵיזֶה רְפוּאָה כַּמָה הָיָה נִכְנַע וּמְשַׁבֵּחַ אוֹתוֹ עַד שֶּלֹא הָיָה יוֹדֵעַ בְּנַפְשׁוֹ בְּאֵיזֶה אוֹפַן לְשַׁלֵם גְּמוּלוֹ, וְכָל שֶׁכֵּן אִם אֶחָד הָיָה מְרַפֵּא אוֹתוֹ חִנָּם לֹא הָיָה נִפְסַק שְׁבָחָיו מִפִּיו עַד עוֹלָם.

וְאַתָּה אָבִינוּ שֶׁבַשָׁמַיִם בְּלִי תַשְׁלוּם גְּמוּל בְּרָאתָנוּ שְּׁלֵמִים בְּאֵבָרֵינוּ, וּבְכָל בוֹקֶר וָבוֹקֶר אַתָּה מַחֲזִיר לָנוּ כּוֹחוֹת הַחֲמִשָׁה חוּשִׁים וְכּוֹחוֹת שְּׁאָר אֵבְרֵי הַגוּף, וְאַף כִּי הִרְבֵּינוּ לְהַכְעִיסֶךָ בְּאֵלוּ הַכּוֹחוֹת אַתָּה בְּרַחֲמֶיךָ הָרַבִּים סוֹבֵל וּמַעֲלִים עַיִן מִסוֹרְרִים וְאַף בְּשָׁעָה שֶׁהָאָדָם עוֹבֵר עֲבֵירָה ח''ו וּבוֹרֵא קְלִיפּוֹת וְחִיצוֹנִיִים הַמַקְצִיפִים אֶת הַבּוֹרֵא, אַף עַל פִּי כֵן אַתָּה מַשְׁפִּיעַ בֹּו כֹּחֲךָ וְהַשְׁפָּעַת חַיֵי נַפְשׁוֹ אַף בִּשְׁעַת עֲשִׂיַת הָעֲבֵירָה חָלִילָה, וְכַמָה גָדְלוּ חֲסָדֶיךָ וְרַחֲמֶיךָ בָזוֹ וְאֵיךְ יַסְפִּיק לָנוּ הַהוֹדָאָה וְהַשֶׁבַח עַל הַחֲסָדִים הַגְּדוֹלִים וַעֲצוּמִים כָּאֵלוּ, וּבִפְרָט עַל חֲסָדִים גְּדוֹלִים וַעֲצוּמִים כָּאֵלוּ שֶׁאַתָּה זָן וּמְפַרְנֵס אוֹתָנוּ בְּכָל יוֹם בְּכָל עֵת וּבְכָל שָׁעָה וְאַתָּה מַצִיל אוֹתָנוּ מִכָּל פִּגְעֵי הָעוֹלָם וְהֵזֵיקָיו (וְעוֹד עָשִׂיתָ עִמָדִי חֲסָדִים גְדוֹלִים שֶׁהִזְמַנְתָּ לִי אִשָׁה יִשְׂרְאֵלִית וְחוֹנַנְתָּנִי בְּבָנִים וּבָנוֹת).

וּבִּפְרָט עַל הַחֶסֶד הַגָּדוֹל וְהַעָצוּם שֶׁלֹא עֲשִׂיתָנוּ כְּגוֹיֵי הַאֲרָצוֹת וְלֹא שַׂמְתָּנוּ כְּמִשְׁפְּחוֹת הַאֲדָמָה, וּבָרָאתָּ אוֹתִי מֵאוּמָה הַקְּדוֹשָׁה עַמְּךָ יִשְׂרָאֵל הַקְּדוֹשִׁים אֲשֶׁר בָּחַרְתָּ אוֹתָם מִכָּל הַאוּמוֹת וְהִרְאִיתָ לָהֶם בְּכָל דוֹר וָדוֹר גוֹדֶל חִיבָּתְךָ וְאַהֲבָתְךָ, וְקֵרַבְתָם לְשִׁמְךָ הַגָדוֹל וְהַקָדוֹשׁ לִפְנַי וְלִפְנִים, וְחִבַּבְתָּם יוֹתֵר מִמַלְאֲכֵי הַשָׁרֵת כְּדְאָמַר ר' יִשְׁמָעאֵל אַשְׁרֵיהֵם יִשְׂרָאֵל כַּמָה הֵם חֲבִיבִים לִפְנֵי הַקָדוֹשׁ בָּרוּךְ הוּא יוֹתֵר מִמַלְאֲכֵי הַשָׂרֵת, כִּי בְּשָׁעָה שֶׁמַלְאָכֵי הַשָׁרֵת מְבַקְשִׁים לוֹמַר שִׁירָה לְפָנָיו יִתְבָּרַךְ וְיִתְרוֹמָם וְיִתְעַלֶה, תְּחִילָה בָּאִים סְבִיבוֹת כִּסֵא הַכָּבוֹד נַהֲרֵי אֵשׁ וּגְבָעוֹת שֶׁל לֶהָבָה וְהַקָדוֹשׁ בָּרוּךְ הוּא אוֹמֵר לָהֶם הֶחֱרִישׁוּ אֶלָא כָּל מַלְאָךְ וּמַלְאָךְ וְכָל חַיָה וְחַיָה וְכָל אוֹפַן וְאוֹפַן שֶׁבָּרָאתִי, עַד שֶׁאֶשְׁמַע וְאַאֲזִין תְּחִלָה קוֹל שִׁירָתָם וְתוּשְׁבַּחְתָּן שֶׁל יִשְׂרָאֵל בָּנַי שֶׁנֶאֱמַר בְּרָן יַחַד כּוֹכְבֵי בוֹקֶר וָיָרִיעוּ כָּל בְּנֵי אֱלֹהִים.

אֵימָתַי כְּבוֹדוֹ בַּשָׁמַיִם כְּדִכְתִיב רוּמָה עַל שָׁמַיִם אֱלֹהִים, בִּזְמַן שֶׁעַל כָּל הָאָרֶץ כְּבוֹדֶךָ, וְאַלְפֵי וְרִבְבוֹת שֶׁל מַלְאֲכֵי הַשָׁרֵת עוֹמְדִין לִימִין הַכִּסֵא וּכְנֶגְדָן לִשְׂמֹאל הַכִּסֵא מַקִיפִים וְעוֹמְדִים שָׂרִים גְּדוֹלִים הַמְמוּנִים עַל כָּל גְדוּד וּגְדוּד וְעַל כָּל צָבָא וְצָבָא מִיָמִין הַכִּסֵא וּמִשְׂמֹאל הַכִּסֵא וּמְלַמְדִין לוֹמַר שִׁירָה וְזִמְרָה לִכְבוֹדוֹ שֶׁל מֶלֶךְ מַלְכֵי הַמְלָכִים הַקָדוֹשׁ בָּרוּךְ הוּא, וּכְשֶׁמַגִיעַ זְמַן לוֹמַר שִׁירָה הַשַׂר הַגָדוֹל הַנִכְבָּד וְהַנוֹרָא (שמעיאל) עוֹמֵד עַל חַלוֹנֵי הַרָקִיעַ הַתַחְתוֹן לִשְׁמוֹעַ וּלְהַקְשִׁיב קוֹל שִׁירוֹת וְתִּשְׁבָּחוֹת הַעוֹלוֹת מִן הָאָרֶץ מִבָּתֵי כְּנֵסִיוֹת וּמִבָּתֵי מִדְרָשׁוֹת לְהַשְׁמִיעַ לִבְנֵי עֲרָבוֹת, וּמִפְּנֵי מַה עוֹשֶׂה כֵּן, שֶׁאֵין רְשׁוּת לְמַלְאֲכֵי הַשָׁרֵת לוֹמַר שִׁירָה עַד שֶׁפּוֹתְחִין יִשְׂרָאֵל תְּחִילָה בְּשִׁירָה שֶׁנֶאֱמַר רוֹמְמוּ ה' אֱלֹהֵינוּ וְהִשְׁתַחֲווּ לְהַר קָדְשׁוֹ אֵלוּ מַלְאֲכֵי הַשָׁרֵת, וְכָל מַלְאֲכֵי רָקִיעַ וְרָקִיעַ כְּשֶׁשׁוֹמְעִין שִׁירוֹת וְתִּשְׁבָּחוֹת שֶׁאוֹמְרִים מִלְמַטָּה הֵם פּוֹתְחִים פִּיהֶם וְאוֹמְרִים קָדוֹשׁ קָדוֹשׁ מִלְמַעְלָה, וְלָמָה נִקְרָא שְׁמוֹ (שמעיאל) מִפְּנֵי שֶׁהוּא עוֹמֵד בְּכָל שַׁעַר וְשַׁעַר וּמַשְׁמִיעַ קוֹל שִׁירוֹת וְתִשְׁבָּחוֹת הַעוֹלוֹת מִן הַאָרֶץ מִתּוֹךְ בָּתֵי כְּנֵסִיוֹת וּבָתֵי מִדְרָשׁוֹת לִפְנֵי כָּל רָקִיעַ וְרָקִיעַ.

ואחר כך יאמר:

ברוך הוא אלהינו שבראנו לכבודו להללו ולשבחו ולספר הודו, מכל אום גבר עלינו חסדו, לכן בכל לב ובכל נפש ובכל מאודו נמליכו וניחדו.

ויקבל עליו עול מלכות שמים באמת ובאמונה:

שמע ישראל ה' אלהינו ה' אחד.

(בלחש):

ברוך שם כבוד מלכותו לעולם ועד.

(ז' פעמים):

ה' הוא האלהים.

(ב' פעמים):

ה' מלך ה' מלך ה' ימלוך לעולם ועד.

ואחר כך לזמר שירי יחוד כגון מספר חרדים או 'אין כאלהינו', 'ויאתיו' וכדומה.
