בָּרוּךְ אַתָּה יְדֹוָד(אדון הכול) אֱלֹהֵינוּ(בעל היכולת שיחד שמו עלינו) מֶלֶךְ הָעוֹלָם עַל: הַמִחְיָה(המזונות המחיים את האדם) וְעַל הַכַּלְכָּלָה (יותר מהמינמום אלא בשפע כמו בשבע כלכלתנו)

וְעַל תְּנוּבַת(צמיחה נקרא תנובה) הַשָּׂדֶה וְעַל אֶרֶץ חֶמְדָּה(שחמדוה עליונים ותחתונים) טוֹבָה (שיש בה מכל טובות שבעולם) וּרְחָבָה (ארץ הצבי שמתרחבת לפי הצורך) שֶׁרָצִיתָ(הקב"ה ***רצה*** להכניס את בניו חביביו לארץ החביבה עליו ולא רק מחמת ההבטחה (ויש לעיין מדוע רק בעל המחיה מזכירים את נתינת הארץ ברצון ולא מזכירים זאת בברהמ"ז? יתכן ובשבעת המינים שבארץ ישראל ניכר יותר הרצון של ה' בעמו שכן לחם חייב היה לתת להם שכן אחרית ימותו אבל פירות מתוקים זו מתנה שמראה על רצונו בהם)הקושיה בחוברת הגרח"ק והתירוץ מה שחנני ה') וְהִנְחַלְתָּ לַאֲבוֹתֵינוּ (הארץ נחלה היא לנו מאבותינו) לֶאֱכֹל מִפִּרְיָהּ וְלִשְׂבּוֹעַ מִטּוּבָהּ(מסביר הב"ח כי באכילת פירותיה אנו *ניזונים מקדושת השכינה* ומטהרתה) רַחֵם (נָא) יְדֹוָד אֱלֹהֵינוּ עַל יִשְׂרָאֵל עַמֶּךָ וְעַל יְרוּשָׁלַיִם עִירֶךָ וְעַל צִיּוֹן מִשְׁכַּן כְּבוֹדֶךָ וְעַל מִזְבְּחֶךָ וְעַל הֵיכָלֶךָ, וּבְנֵה יְרוּשָׁלַיִם עִיר הַקֹּדֶשׁ בִּמְהֵרָה בְּיָמֵינוּ וְהַעֲלֵנוּ לְתוֹכָהּ(חוזר על ארץ ישראל ה.ג.) וְשַׂמְּחֵנוּ בְּבִנְיָנָהּ וְנֹאכַל מִפִּרְיָּהּ וְנִשְׂבַּע מִטּוּבָהּ(של ארץ ישראל) וּנְבָרֶכְךָ עָלֶיהָ בִּקְדֻשָּׁה וּבְטָהֳרָה(היות ונהיה בארץ קדושה וטהורה מטומאת ארץ העמים ממילא הברכות יהיו בקדושה ובטהרה.ה.ג. )

בשבת - וּרְצֵה וְהַחֲלִיצֵנוּ בְּיוֹם הַשַּׁבָּת הַזֶּה:

בר"ח - וְזָכְרֵנוּ לְטוֹבָה בְּיוֹם רֹאשׁ הַחֹדֶשׁ הַזֶּה:

בר"ה - וְזָכְרֵנוּ לְטוֹבָה בְּיוֹם הַזִּכָּרוֹן הַזֶּה:

ביו"ט - וְשַׂמְחֵנוּ בְּיוֹם חַג (פלוני) הַזֶּה:

כִּי אַתָּה יְדֹוָד טוֹב וּמֵטִיב לַכֹּל וְנוֹדֶה לְךָ עַל הָאָרֶץ וְעַל:
